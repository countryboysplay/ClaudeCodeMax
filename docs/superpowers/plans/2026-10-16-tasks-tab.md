# Tasks Tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Tasks tab that shows, per project, each request, the files Claude changed, Claude's closing note, unfinished or unexplained-skip badges, and every deferred-work comment, flagging any with no reason.

**Architecture:** Two pure main-process modules read data already on disk: `tasks.ts` parses Claude Code transcripts and `deferred.ts` scans project source for `shortcut:`/`ponytail:`/`TODO`/`FIXME` comments. One `tasks:get` IPC call returns both to a new `Tasks.tsx` panel. No model calls, nothing added to Claude's context.

**Tech Stack:** Electron main (Node fs, git via the existing `git()` helper), React renderer, Vitest, Playwright.

**Spec:** `docs/superpowers/specs/2026-10-16-tasks-tab-design.md`

## Global Constraints

- Zero tokens: no model calls, no hooks, nothing injected into sessions.
- Transcripts: `~/.claude/projects/<slugFor(project)>/*.jsonl` (`slugFor` from `src/main/memory/files.ts`; the path override is `claudeProjectsDir()`).
- At most 200 tasks per project, newest first. `note` is at most 600 characters.
- Deferred scan: `git ls-files` when available, otherwise a walk that skips `node_modules`, `.git`, `dist`, `out`, `build` and `.venv`. Skip files over 1 MB and binary files. Stop at 20,000 files and report it.
- Transcript folders reach 114 MB on this machine, so read them asynchronously, one file at a time.
- Follow the existing styles, keyboard path and WCAG AA contrast rules.

## Review Focus

1. **A tool result or local-command line in the transcript must not start a new task.** Pinned in Task 2's test (`tool_result` and `<local-command-stdout>` lines).
2. **An interrupt recorded inside a tool-result message** (`[Request interrupted by user for tool use]`) must still mark the task unfinished. Pinned in Task 2.
3. **The same mid-turn message enqueued twice** must show once. Pinned in Task 2.
4. **A project with no transcripts, or a non-git folder,** must render an empty state, not throw. Pinned in Tasks 1 and 2.
5. **Opening a file must only open files inside the project.** A crafted path like `../../x` must be refused. Pinned in Task 3's `isInside` test.

```ts
import { readFileSync, readdirSync, statSync } from 'node:fs'
import { join } from 'node:path'
import { git } from './memory/store'
import type { Deferred } from '../shared/types'

const MAX_FILES = 20_000
const MAX_BYTES = 1 << 20
const SKIP_DIRS = new Set(['node_modules', '.git', 'dist', 'out', 'build', '.venv'])
// A marker only counts after a comment opener, so prose or strings that mention TODO aren't listed.
const MARK = /(?:\/\/|#|\/\*|<!--|--|^\s*\*)\s*(shortcut:|ponytail:|TODO\b|FIXME\b):?\s*(.*)$/

export const hasReason = (text: string): boolean =>
  /, | — | - /.test(text) || /\b(because|since|until|so that|to avoid|when)\b/i.test(text)

async function listFiles(dir: string): Promise<string[]> {
  try {
    return (await git(dir, 'ls-files', '-z', '--cached', '--others', '--exclude-standard')).split('\0').filter(Boolean)
  } catch {
    // not a git repo, or git missing: walk the folder instead
    const out: string[] = []
    const walk = (rel: string): void => {
      for (const d of readdirSync(join(dir, rel), { withFileTypes: true })) {
        if (out.length > MAX_FILES) return
        const p = rel ? `${rel}/${d.name}` : d.name
        if (d.isDirectory()) {
          if (!SKIP_DIRS.has(d.name)) walk(p)
        } else if (d.isFile()) out.push(p)
      }
    }
    walk('')
    return out
  }
}

export async function scanDeferred(dir: string): Promise<{ deferred: Deferred[]; scanCapped: boolean }> {
  // git ls-files --others lists untracked node_modules etc. unless they're gitignored, so filter both ways.
  const files = (await listFiles(dir)).filter(f => !f.split(/[\\/]/).some(p => SKIP_DIRS.has(p)))
  const deferred: Deferred[] = []
  for (const file of files.slice(0, MAX_FILES)) {
    let buf: Buffer
    try {
      const full = join(dir, file)
      if (statSync(full).size > MAX_BYTES) continue
      buf = readFileSync(full)
    } catch {
      continue // deleted since it was listed, or unreadable
    }
    if (buf.subarray(0, 8000).includes(0)) continue // binary
    buf
      .toString('utf8')
      .split(/\r?\n/)
      .forEach((line, i) => {
        const m = MARK.exec(line)
        if (!m) return
        const text = m[2].replace(/\s*(\*\/|-->)\s*$/, '').trim()
        deferred.push({ file: file.replace(/\\/g, '/'), line: i + 1, marker: m[1].replace(/:$/, ''), text, reason: hasReason(text) })
      })
  }
  // Stable sort: items with no reason first, otherwise in file order.
  deferred.sort((a, b) => Number(a.reason) - Number(b.reason))
  return { deferred, scanCapped: files.length > MAX_FILES }
}
```

```ts
import { existsSync, readdirSync, readFileSync } from 'node:fs'
import { readFile } from 'node:fs/promises'
import { basename, isAbsolute, join, relative } from 'node:path'
import { slugFor } from './memory/files'
import { hasReason } from './deferred'
import type { Task } from '../shared/types'

const MAX_TASKS = 200
const EDIT_TOOLS = new Set(['Write', 'Edit', 'NotebookEdit'])
const SKIP = /\b(skip|skipped|skipping|didn't|did not|not checked|deferred|later|left out|out of scope)\b/i
const INTERRUPT = '[Request interrupted'

const lastParagraph = (s: string) => (s.trim().split(/\n\s*\n/).pop() ?? '').trim().slice(0, 600)

// The request the user typed, or null for user-role lines that aren't one (tool results, command output).
function typed(content: unknown): string | null {
  const text =
    typeof content === 'string'
      ? content
      : Array.isArray(content) && !content.some(b => b?.type === 'tool_result')
        ? content.filter(b => b?.type === 'text').map(b => String(b.text)).join('\n')
        : ''
  if (!text.trim()) return null
  const cmd = /<command-name>([^<]*)<\/command-name>/.exec(text)
  if (cmd) return cmd[1].trim()
  return text.trimStart().startsWith('<') ? null : text.trim()
}

export function parseTranscript(text: string, rel: (file: string) => string = f => f): Task[] {
  const tasks: Task[] = []
  let t: Task | null = null
  for (const raw of text.split('\n')) {
    let o: any
    try {
      o = JSON.parse(raw)
    } catch {
      continue
    }
    if (o.isMeta || o.isSidechain) continue
    if (o.type === 'queue-operation') {
      if (t && o.operation === 'enqueue' && typeof o.content === 'string' && !t.added.includes(o.content)) t.added.push(o.content)
    } else if (o.type === 'user') {
      if (JSON.stringify(o.message?.content ?? '').includes(INTERRUPT)) {
        if (t) t.unfinished = true
        continue
      }
      const asked = typed(o.message?.content)
      if (asked === null) continue
      t = { when: String(o.timestamp ?? ''), asked, added: [], files: [], note: '', reply: '', unfinished: false, skipFlag: false, checklist: [] }
      tasks.push(t)
    } else if (o.type === 'assistant' && t && Array.isArray(o.message?.content)) {
      for (const b of o.message.content) {
        if (b?.type === 'text' && String(b.text).trim()) t.reply = String(b.text)
        const f = b?.input?.file_path ?? b?.input?.notebook_path
        if (b?.type === 'tool_use' && EDIT_TOOLS.has(b.name) && typeof f === 'string' && !t.files.includes(rel(f))) t.files.push(rel(f))
      }
    }
  }
  for (const x of tasks) {
    x.note = lastParagraph(x.reply)
    if (!x.reply) x.unfinished = true
    x.skipFlag = SKIP.test(x.note) && !hasReason(x.note)
  }
  return tasks
}

function readChecklist(dir: string): Task['checklist'] {
  if (!existsSync(dir)) return []
  return readdirSync(dir)
    .filter(f => f.endsWith('.json'))
    .sort((a, b) => parseInt(a) - parseInt(b))
    .flatMap(f => {
      try {
        const j = JSON.parse(readFileSync(join(dir, f), 'utf8'))
        return typeof j.subject === 'string' ? [{ subject: j.subject, status: String(j.status ?? '') }] : []
      } catch {
        return []
      }
    })
}

// Reads one transcript at a time so a large project (100+ MB) doesn't freeze the window.
export async function readTasks(projectDir: string, claudeProjects: string, claudeTasks: string): Promise<Task[]> {
  const dir = join(claudeProjects, slugFor(projectDir))
  if (!existsSync(dir)) return []
  const rel = (f: string) => {
    const r = relative(projectDir, f)
    return r && !r.startsWith('..') && !isAbsolute(r) ? r.replace(/\\/g, '/') : f
  }
  const all: Task[] = []
  for (const f of readdirSync(dir).filter(n => n.endsWith('.jsonl'))) {
    let text: string
    try {
      text = await readFile(join(dir, f), 'utf8')
    } catch {
      continue
    }
    const tasks = parseTranscript(text, rel)
    // Task files are per session, so the checklist goes on the session's last task.
    const last = tasks.at(-1)
    if (last) last.checklist = readChecklist(join(claudeTasks, basename(f, '.jsonl')))
    all.push(...tasks)
  }
  return all.sort((a, b) => b.when.localeCompare(a.when)).slice(0, MAX_TASKS)
}
```

```ts
import { isAbsolute, relative, resolve } from 'node:path'

// True when `file` (relative to the project, or absolute) is inside `projectDir`.
export function isInside(projectDir: string, file: string): boolean {
  if (!file) return false
  const r = relative(projectDir, resolve(projectDir, file))
  return r !== '' && !r.startsWith('..') && !isAbsolute(r)
}
```

```tsx
      )}
      <p className="muted">Files changed through Bash or PowerShell commands aren't listed.</p>
    </div>
  )
}
```
